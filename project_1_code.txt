# Extract mid signal ±window around TSS 
#-------------------------
def extract_mean_signal(bw_path, gene_df, window=2000):
    bw = pyBigWig.open(bw_path)
    signals = []
    for _, row in gene_df.iterrows():
        chrom, start, end = row["chr"], int(row["TSS_start"]), int(row["TSS_end"])
        tss_center = (start + end) // 2
        start = max(0, tss_center - window)
        end = tss_center + window
        try:
            values = np.array(bw.values(chrom, start, end))
            mean_signal = np.nanmean(values)
        except Exception:
            mean_signal = np.nan
        signals.append(mean_signal)
    bw.close()
    return signals

def extract_bed_features(bed_path, gene_df, window=2000):
    
    bed_df = pd.read_csv(bed_path, sep="\t", header=None,
                         names=["chr","start","end","peak_id","score",".","val1","val2","val3","val4"])
    
    features = {"peak_count_window":[], "peak_max_score":[], "peak_sum_len":[], "nearest_peak_dist":[]}
    
    for _, row in gene_df.iterrows():
        chrom, tss_center = row["chr"], (row["TSS_start"]+row["TSS_end"])//2
        start_window = tss_center - window
        end_window   = tss_center + window
        
        chrom_peaks = bed_df[bed_df["chr"] == chrom]
        overlaps = chrom_peaks[(chrom_peaks["end"] >= start_window) & (chrom_peaks["start"] <= end_window)]
        
        features["peak_count_window"].append(len(overlaps))
        
        if len(overlaps)>0:
            features["peak_max_score"].append(overlaps["score"].max())
            features["peak_sum_len"].append((overlaps["end"]-overlaps["start"]).sum())
            dists = np.minimum(np.abs(overlaps["start"]-tss_center), np.abs(overlaps["end"]-tss_center))
            features["nearest_peak_dist"].append(dists.min())
        else:
            features["peak_max_score"].append(0)
            features["peak_sum_len"].append(0)
            features["nearest_peak_dist"].append(window)
            
    return pd.DataFrame(features)

    val_info   = pd.read_csv(info_path_val, sep="\t")
    
    y_train = pd.read_csv(y_path_train, sep="\t")["gex"]
    y_val   = pd.read_csv(y_path_val, sep="\t")["gex"]

    def build_feature_matrix(info_df, dataset_name=""):
        df = pd.DataFrame({"gene_name": info_df["gene_name"]})
        
        for mark, folder in feature_paths.items():
            bw_file = os.path.join(folder, f"{dataset}.bw")
            print(f"  → {mark} (BigWig) [{dataset_name}] ...")
            df[mark] = extract_mean_signal(bw_file, info_df)
            
        for mark, folder in bed_paths.items():
            bed_file = os.path.join(folder, f"{dataset}.bed")
            print(f"  → {mark} (BED) [{dataset_name}] ...")
            bed_feat = extract_bed_features(bed_file, info_df)

            df = pd.concat([df, bed_feat], axis=1)
    
        return df

    X_val   = build_feature_matrix(val_info, dataset_name="val")
    
    return X_train, y_train, X_val, y_val
